# The device serves the phone app itself, so the files in ../../app are built into the firmware.
set(app "${CMAKE_CURRENT_LIST_DIR}/../../app")

idf_component_register(
    SRCS "main.c"
         "door.c"
         "intercom.c"
         "knock.c"
         "board.c"
         "events.c"
         "keys.c"
         "settings.c"
         "api.c"
         "web.c"
         "remote.c"
         "notify.c"
         "net.c"
         "captive_dns.c"
         "util.c"
    INCLUDE_DIRS "."
    EMBED_TXTFILES "${app}/index.html"
                   "${app}/style.css"
                   "${app}/main.js"
                   "${app}/device.js"
                   "${app}/crypto.js"
                   "${app}/mqtt.js"
                   "${app}/sw.js"
                   "${app}/manifest.webmanifest"
                   "${app}/icon.svg"
    EMBED_FILES "${app}/icon-180.png"
                "${app}/icon-192.png"
                "${app}/icon-512.png"
                "${app}/icon-maskable.png"
)
